**Goal:** Claude Code와 Codex CLI를 번갈아 쓸 때, 갈아타는 순간 ≤900바이트의 표식이 자동으로 다음 세션에 주입되고, 깊은 내용은 원본 트랜스크립트를 하드링크+오프셋으로 가리켜 조회하게 만든다.

**Architecture:** 훅으로 "지금 저장"하지 않는다. 각 세션 시작 때 원장에 한 줄을 남기고, 다음 세션 시작 때 상대 하네스의 로그를 지연 소급 수집해 산출물을 만든다. 산출물 생성 지점은 `mint.py` 하나이고 예산 단정이 함수의 마지막 문장이다. 아카이브는 재직렬화하지 않고 `os.link()` 하드링크 + TSV 오프셋 색인으로만 존재한다. LLM 호출은 어디에도 없다.

**Tech Stack:** Python 3.9.25, 표준 라이브러리만. 의존성 0. 테스트는 `unittest`(stdlib). 셸은 bash.

**Spec:** `docs/superpowers/specs/2026-09-22-omhc-design.md`

## Global Constraints

- Python **3.9** 호환. `@dataclass(slots=True)`는 이 머신에 **없다**(TypeError). `frozen=True`만 쓴다 — 3.9에서 미선언 속성 할당도 `FrozenInstanceError`로 막히는 것이 확인됐다.
- 모듈 첫 줄은 `from __future__ import annotations` (PEP 585/604 표기를 3.9에서 쓰기 위해).
- 표준 라이브러리 외 **의존성 금지**. `pytest` 대신 `python3 -m unittest`.
- **LLM 호출 금지.** 어떤 모듈도 서브프로세스로 모델을 부르지 않는다.
- 어떤 IR 레코드에도 **툴 이름이 들어갈 필드를 두지 않는다**. 동사는 닫힌 집합.
- `Event`의 화자 필드는 `author`이고 값은 `"human" | "agent" | "harness"` **셋뿐**이다. `"human"`만 축자 중계된다.
- 주입 예산은 **900 바이트**(UTF-8 인코딩 기준). `mint()`의 마지막 문장이 `assert`.
- 상태 루트는 `~/.omhc/`, 레포 내 바닥은 `<repo>/.omhc/outbox/`. 관리 구간 마커는 `<!-- omhc:begin -->` / `<!-- omhc:end -->`.
- 차단목록(denylist) 항목에 **백슬래시를 넣지 않는다**. 모든 항목은 픽스처 원문에서 목격돼야 한다.
- 훅 경로(`omhc brief`)는 **절대 예외를 던지지 않는다**. 실패 시 빈 stdout + exit 0.
- 글롭은 **깊이 1만**. 중첩 `<session>/subagents/**` 는 읽지 않는다.

# omhc/locate.py
from __future__ import annotations

import hashlib
import os
import subprocess
from typing import Optional


def resolve_repo_root(start: Optional[str] = None) -> str:
    """레포 루트의 THE 정의. git toplevel, 없으면 realpath(cwd).

    호출 지점마다 다르게 정의하면 서브디렉터리에서 세션이 조용히 0건이 된다.
    """
    base = os.path.realpath(start or os.getcwd())
    try:
        out = subprocess.run(
            ["git", "-C", base, "rev-parse", "--show-toplevel"],
            capture_output=True, text=True, timeout=5,
        )
        if out.returncode == 0 and out.stdout.strip():
            return os.path.realpath(out.stdout.strip())
    except (OSError, subprocess.SubprocessError):
        pass
    return base


def repo_key(repo_root: str) -> str:
    """사람이 읽을 수 있는 basename + 경로 해시. 다른 경로의 동명 레포를 구분한다."""
    digest = hashlib.sha1(repo_root.encode("utf-8")).hexdigest()[:8]
    return "{}-{}".format(os.path.basename(repo_root.rstrip("/")), digest)


def is_within(repo_root: str, candidate: str) -> bool:
    """equal-or-descendant. list_sessions의 cwd 일치 규칙."""
    root = os.path.realpath(repo_root).rstrip("/")
    cand = os.path.realpath(candidate).rstrip("/")
    return cand == root or cand.startswith(root + "/")


def relativize(repo_root: str, path: str) -> Optional[str]:
    """절대경로 → 레포 상대 POSIX 경로. 레포 밖이면 None."""
    if not is_within(repo_root, path):
        return None
    root = os.path.realpath(repo_root).rstrip("/")
    rel = os.path.realpath(path)[len(root):].lstrip("/")
    return rel or "."


def state_dir(key: str, home: Optional[str] = None) -> str:
    return os.path.join(home or os.path.expanduser("~"), ".omhc", key)
```

        self.home = self.tmp.name

        rows = ledger.read(home=self.home)
        self.assertEqual(len(rows), 1)
        self.assertEqual(rows[0]["harness"], "claude")

        ledger.append({"repo": "r" * 500, "harness": "claude", "event": "start"}, home=self.home)
        path = os.path.join(self.home, ".omhc", ledger.LEDGER_NAME)
        with open(path, "rb") as fh:
            self.assertLessEqual(len(fh.readline()), 400)

        pids = []
        for _ in range(8):
            pid = os.fork()
            if pid == 0:
                for i in range(200):
                    ledger.append({"repo": "r", "harness": "h", "event": "start", "n": i},
                                  home=self.home)
                os._exit(0)
            pids.append(pid)
        for pid in pids:
            os.waitpid(pid, 0)
        path = os.path.join(self.home, ".omhc", ledger.LEDGER_NAME)
        with open(path) as fh:
            lines = fh.read().splitlines()
        self.assertEqual(len(lines), 1600)
        for line in lines:
            json.loads(line)   # 부분 레코드가 하나라도 있으면 여기서 터진다

    def test_corrupt_half_line_is_skipped_not_fatal(self):
        path = os.path.join(self.home, ".omhc", ledger.LEDGER_NAME)
        os.makedirs(os.path.dirname(path), exist_ok=True)

            fh.write('{"repo":"a","event":"start"}\n{"repo":"b","ev\n{"repo":"c","event":"start"}\n')

# omhc/ledger.py
from __future__ import annotations

import json
import os
from typing import Optional

LEDGER_NAME = "ledger.jsonl"
MAX_LINE = 400


def _path(home: Optional[str]) -> str:
    return os.path.join(home or os.path.expanduser("~"), ".omhc", LEDGER_NAME)


def append(record: dict, home: Optional[str] = None) -> None:
    """한 줄을 O_APPEND 단일 write(2)로 쓴다.

    PIPE_BUF(4096) 이하 단일 write는 O_APPEND에서 원자적이므로 여러 세션이
    동시에 써도 부분 레코드가 생기지 않는다. 400바이트 상한이 그 보장의 근거다.
    """
    line = json.dumps(record, ensure_ascii=False, separators=(",", ":"))
    if len(line.encode("utf-8")) + 1 > MAX_LINE:
        trimmed = dict(record)
        for key in ("path", "cwd", "repo"):
            if key in trimmed and isinstance(trimmed[key], str):
                trimmed[key] = trimmed[key][:80]
        line = json.dumps(trimmed, ensure_ascii=False, separators=(",", ":"))
        line = line.encode("utf-8")[: MAX_LINE - 1].decode("utf-8", "ignore")
    path = _path(home)
    os.makedirs(os.path.dirname(path), exist_ok=True)
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_APPEND, 0o600)
    try:
        os.write(fd, (line + "\n").encode("utf-8"))
    finally:
        os.close(fd)


def read(limit: int = 2000, home: Optional[str] = None) -> list:
    """마지막 limit 줄을 파싱한다. 깨진 줄은 건너뛴다(fail-open)."""
    path = _path(home)
    try:
        with open(path, encoding="utf-8", errors="replace") as fh:
            lines = fh.read().splitlines()
    except OSError:
        return []
    out = []
    for line in lines[-limit:]:
        try:
            row = json.loads(line)
        except ValueError:
            continue
        if isinstance(row, dict):
            out.append(row)
    return out


def newest(repo_key: str, harness: Optional[str] = None,
           exclude_harness: Optional[str] = None, event: str = "start",
           home: Optional[str] = None) -> Optional[dict]:
    """조건에 맞는 가장 최근 줄. 순서는 파일 내 위치이며 타임스탬프가 아니다."""
    for row in reversed(read(home=home)):
        if row.get("repo") != repo_key:
            continue
        if event and row.get("event") != event:
            continue
        if harness and row.get("harness") != harness:
            continue
        if exclude_harness and row.get("harness") == exclude_harness:
            continue
        return row
    return None
```

import dataclasses, unittest
from omhc.event import AUTHORS, VERBS, Event, tally

def mk(**kw):
    base = dict(seq=1, epoch=1.0, author="human", verb="said", ok=True,
                text="hi", arg="", paths=(), offset=0, length=2)
    base.update(kw)
    return Event(**base)

class TestEvent(unittest.TestCase):
    def test_verbs_are_closed_and_neutral(self):
        self.assertEqual(VERBS, frozenset(
            {"said", "inspected", "modified", "ran", "delegated", "researched"}))

    def test_authors_are_exactly_three(self):
        self.assertEqual(AUTHORS, frozenset({"human", "agent", "harness"}))

    def test_vendor_tool_names_are_not_valid_verbs(self):
        for name in ("Bash", "Read", "Edit", "Task", "apply_patch", "shell", "update_plan"):
            with self.assertRaises(ValueError):
                mk(verb=name)

    def test_unknown_author_rejected(self):
        with self.assertRaises(ValueError):
            mk(author="user")

    def test_event_has_no_field_that_could_hold_a_tool_name(self):
        names = {f.name for f in dataclasses.fields(Event)}
        for forbidden in ("tool", "tool_name", "extra", "raw", "metadata", "payload"):
            self.assertNotIn(forbidden, names)

# omhc/event.py
from __future__ import annotations

import collections
from dataclasses import dataclass, field, fields
from typing import Tuple

IR_VERSION = 1

# 닫힌 중립 동사. 툴 어휘 교집합이 공집합이므로(CC: Read/Edit/Bash/Task,
# Codex: shell/apply_patch/update_plan) 벤더 이름을 담을 자리를 두지 않는다.
VERBS = frozenset({"said", "inspected", "modified", "ran", "delegated", "researched"})

# 3값 판별자. "human"만 축자 중계된다. 서브에이전트가 만든 user 레코드를
# 사람의 말로 오판해 중계하는 것을 구조적으로 막는 유일한 장치다.
AUTHORS = frozenset({"human", "agent", "harness"})

# slots=True 는 이 머신의 Python 3.9.25 에 없다. frozen=True 단독으로
# 미선언 속성 할당까지 FrozenInstanceError 로 막히는 것이 확인됐다.
@dataclass(frozen=True)
class Event:
    seq: int
    epoch: float
    author: str
    verb: str
    ok: bool
    text: str
    arg: str
    paths: Tuple[str, ...]
    offset: int
    length: int

    def __post_init__(self) -> None:
        if self.author not in AUTHORS:
            raise ValueError("author must be one of {}: {!r}".format(sorted(AUTHORS), self.author))
        if self.verb not in VERBS:
            raise ValueError("verb must be one of {}: {!r}".format(sorted(VERBS), self.verb))


def tally(events) -> dict:
    by_verb = collections.Counter()
    by_author = collections.Counter()
    failures = 0
    total = 0
    for ev in events:
        total += 1
        by_verb[ev.verb] += 1
        by_author[ev.author] += 1
        if not ev.ok:
            failures += 1
    return {"total": total, "by_verb": dict(by_verb),
            "by_author": dict(by_author), "failures": failures}

# tests/harvest.py
"""실물 세션을 픽스처로 수확하고 골든 카운트를 독립 계산한다.

어댑터를 부르지 않는다. 골든을 피검증 코드로 만들면 순환 검증이 된다.
"""
from __future__ import annotations

import collections
import json
import os
import re
import shutil
import sys

HOME = os.path.expanduser("~")
REPO = "/home/ec2-user/capstone/oh-my-harness-cowork"
FIX = os.path.join(os.path.dirname(os.path.abspath(__file__)), "fixtures")
SLUG = re.sub(r"[^a-zA-Z0-9]", "-", REPO)
CLAUDE_DIR = os.path.join(HOME, ".claude", "projects", SLUG)

TAG = re.compile(r"^<([a-zA-Z][\w:-]*)(\s[^>]*)?>.*?</\1>\s*", re.S)
SYNTHETIC = ("[Request interrupted by user]",)


def is_envelope(text: str) -> bool:
    """내용 전체가 최상위 XML 봉투로만 이뤄졌는가. 태그 이름 목록이 아니라 구조로 판정한다."""
    rest = text.strip()
    if not rest.startswith("<"):
        return False
    while rest:
        m = TAG.match(rest)
        if not m:
            return False
        rest = rest[m.end():].strip()
    return True


def human_text(row: dict):
    if row.get("type") != "user" or row.get("isMeta") or row.get("isSidechain"):
        return None
    content = (row.get("message") or {}).get("content")
    if isinstance(content, str):
        text = content
    elif isinstance(content, list):
        kinds = {b.get("type") for b in content if isinstance(b, dict)}
        if kinds != {"text"}:
            return None
        text = "".join(b.get("text", "") for b in content if b.get("type") == "text")
    else:
        return None
    if is_envelope(text) or text.strip() in SYNTHETIC:
        return None
    return text


def main() -> int:
    os.makedirs(os.path.join(FIX, "claude"), exist_ok=True)
    os.makedirs(os.path.join(FIX, "codex"), exist_ok=True)

    tops = sorted((os.path.join(CLAUDE_DIR, n) for n in os.listdir(CLAUDE_DIR)
                   if n.endswith(".jsonl")), key=os.path.getsize, reverse=True)
    src = tops[0]
    shutil.copyfile(src, os.path.join(FIX, "claude", "live.jsonl"))

    types = collections.Counter()
    atts = collections.Counter()
    first_cwd, no_cwd, humans = None, 0, 0
    for i, line in enumerate(open(src, encoding="utf-8", errors="replace")):
        try:
            row = json.loads(line)
        except ValueError:
            continue
        types[str(row.get("type"))] += 1
        if row.get("type") == "attachment":
            atts[str((row.get("attachment") or {}).get("type"))] += 1
        if row.get("cwd"):
            if first_cwd is None:
                first_cwd = i
        else:
            no_cwd += 1
        if human_text(row) is not None:
            humans += 1

    sess_id = os.path.basename(src)[: -len(".jsonl")]
    sub_root = os.path.join(CLAUDE_DIR, sess_id, "subagents")
    sub = None
    for dirpath, _dirs, names in os.walk(sub_root):
        for name in names:
            if name.endswith(".jsonl"):
                sub = os.path.join(dirpath, name)
                break
        if sub:
            break
    if sub:
        shutil.copyfile(sub, os.path.join(FIX, "claude", "subagent.jsonl"))

    codex_root = os.path.join(HOME, ".codex", "sessions")
    rollouts = []
    for dirpath, _dirs, names in os.walk(codex_root):
        for name in names:
            if name.startswith("rollout-") and name.endswith(".jsonl"):
                rollouts.append(os.path.join(dirpath, name))
    rollouts.sort(key=os.path.getmtime, reverse=True)
    ctypes = collections.Counter()
    cwd = None
    if rollouts:
        shutil.copyfile(rollouts[0], os.path.join(FIX, "codex", "exec.jsonl"))
        for line in open(rollouts[0], encoding="utf-8", errors="replace"):
            try:
                row = json.loads(line)
            except ValueError:
                continue
            kind = str(row.get("type"))
            if kind == "response_item":
                kind += ":" + str((row.get("payload") or {}).get("type"))
            ctypes[kind] += 1
            if row.get("type") == "session_meta":
                cwd = (row.get("payload") or {}).get("cwd")

    expected = {
        "claude": {"records": sum(types.values()), "types": dict(types),
                   "first_cwd_index": first_cwd, "no_cwd": no_cwd,
                   "human_turns": humans, "attachment_subtypes": dict(atts)},
        "codex": {"records": sum(ctypes.values()), "types": dict(ctypes), "cwd": cwd},
    }
    with open(os.path.join(FIX, "expected.json"), "w", encoding="utf-8") as fh:
        json.dump(expected, fh, ensure_ascii=False, indent=2, sort_keys=True)
    print(json.dumps(expected["claude"]["types"], ensure_ascii=False))
    print("human_turns =", humans, "records =", sum(types.values()))
    return 0


if __name__ == "__main__":
    sys.exit(main())
